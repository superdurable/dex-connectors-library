const (
	credentialIsolationReleaseEnvironment = "DEX_CONNECTOR_COMPAT_CREDENTIAL_ISOLATION_RELEASE"
	credentialIsolationConnectionName     = "credential-isolation-keys"
	credentialIsolationFrameProtocol      = "0.2.0"
)

// credentialIsolationUnusableValues are the stored forms of a field that must send nothing.
var credentialIsolationUnusableValues = []credentialIsolationUnusableValue{
	{name: "missing", isOmitted: true},
	{name: "blank", value: ""},
	{name: "whitespace", value: "   "},
}

// credentialIsolationProviderFailures run Dex Web's provider failure path after it has loaded a key.
var credentialIsolationProviderFailures = []credentialIsolationProviderFailure{
	{name: "401 echoing Authorization", behavior: credentialIsolationProviderRejectsEchoingAuthorization},
	{name: "2xx echoing the key", behavior: credentialIsolationProviderListsModelsEchoingKey},
	{name: "closed connection", behavior: credentialIsolationProviderClosesConnection},
}

// TestStudioCommandCredentialIsolation runs a two-key fixture's Studio commands
// through Dex Web configured as dexcli dev configures it.
func TestStudioCommandCredentialIsolation(t *testing.T) {
	scenario := newCredentialIsolationScenario(t)
	scenario.requireConnectionStatus(t, "Missing")
	scenario.saveCredentials(t, scenario.allCredentials())
	scenario.requireConnectionStatus(t, "Ready")
	scenario.openStudioSession(t)
	for _, command := range scenario.commands {
		t.Run(command.id+" reaches only its own host", func(t *testing.T) {
			scenario.requireCommandReachesOnlyItsHost(t, command)
		})
	}
	for _, command := range scenario.commands {
		for _, failure := range credentialIsolationProviderFailures {
			t.Run(command.id+" with provider "+failure.name+" returns neither key", func(t *testing.T) {
				scenario.requireProviderFailureReturnsNeitherKey(t, command, failure)
			})
		}
	}
	for _, command := range scenario.commands {
		for _, unusable := range credentialIsolationUnusableValues {
			t.Run(command.id+" with "+unusable.name+" "+command.field+" sends nothing", func(t *testing.T) {
				scenario.requireUnusableFieldSendsNothing(t, command, unusable)
			})
		}
	}
}

type credentialIsolationScenario struct {
	release            credentialIsolationRelease
	commands           []credentialIsolationCommand
	secrets            []credentialIsolationSecret
	network            *credentialIsolationProviderNetwork
	webServer          *httptest.Server
	csrfToken          string
	definitionRevision string
	connection         credentialIsolationConnectionView
	sessionNonce       string
	frameRequestCount  int
}

type credentialIsolationUnusableValue struct {
	name      string
	isOmitted bool
	value     string
}

type credentialIsolationProviderFailure struct {
	name     string
	behavior credentialIsolationProviderBehavior
}

type credentialIsolationProviderBehavior int

const (
	credentialIsolationProviderListsModels credentialIsolationProviderBehavior = iota
	credentialIsolationProviderRejectsEchoingAuthorization
	credentialIsolationProviderListsModelsEchoingKey
	credentialIsolationProviderClosesConnection
)

// credentialIsolationRelease is the connector-release.json subset the scenario reads.
type credentialIsolationRelease struct {
	ConnectorID string `json:"connectorId"`
	ModulePath  string `json:"modulePath"`
	Version     string `json:"version"`
	Manifest    struct {
		Spec struct {
			Provider string `json:"provider"`
			Auth     struct {
				Fields []struct {
					Name     string `json:"name"`
					Type     string `json:"type"`
					Required bool   `json:"required"`
				} `json:"fields"`
			} `json:"auth"`
			Studio *struct {
				Setup struct {
					BackendCapabilities []string `json:"backendCapabilities"`
				} `json:"setup"`
				Commands []struct {
					ID         string `json:"id"`
					Capability string `json:"capability"`
					Request    struct {
						Method     string `json:"method"`
						URL        string `json:"url"`
						Credential struct {
							Field  string `json:"field"`
							Scheme string `json:"scheme"`
						} `json:"credential"`
						FixedQuery   map[string]string `json:"fixedQuery"`
						FixedHeaders map[string]string `json:"fixedHeaders"`
						Parameters   []json.RawMessage `json:"parameters"`
					} `json:"request"`
				} `json:"commands"`
				Units []struct {
					ID      string `json:"id"`
					Outputs []struct {
						Name string `json:"name"`
					} `json:"outputs"`
				} `json:"units"`
			} `json:"studio"`
			Operations []struct {
				Name string `json:"name"`
				Kind string `json:"kind"`
			} `json:"operations"`
		} `json:"spec"`
	} `json:"manifest"`
}

type credentialIsolationCommand struct {
	id         string
	capability string
	field      string
	target     *url.URL
	fixedQuery url.Values
}

type credentialIsolationSecret struct {
	field string
	value string
	// marker is the random part of value; any occurrence of it is a leak.
	marker string
}

type credentialIsolationConnectionView struct {
	ConnectorID    string `json:"connectorId"`
	ConnectionName string `json:"connectionName"`
	Status         string `json:"status"`
	LocalOverride  bool   `json:"localOverride"`
	Uses           []struct {
		FlowName        string                         `json:"flowName"`
		StepName        string                         `json:"stepName"`
		OperationID     string                         `json:"operationId"`
		ConfigurationUI api.V2ConnectorConfigurationUI `json:"configurationUI"`
		Configuration   map[string]json.RawMessage     `json:"configuration"`
	} `json:"uses"`
}

func newCredentialIsolationScenario(t *testing.T) *credentialIsolationScenario {
	t.Helper()
	releaseDirectory := os.Getenv(credentialIsolationReleaseEnvironment)
	if releaseDirectory == "" {
		t.Fatal(credentialIsolationReleaseEnvironment + " is required")
	}
	metadata, err := os.ReadFile(filepath.Join(releaseDirectory, connectorReleaseMetadataName))
	if err != nil {
		t.Fatal(err)
	}
	scenario := &credentialIsolationScenario{}
	if err := json.Unmarshal(metadata, &scenario.release); err != nil {
		t.Fatalf("decode credential isolation release: %v", err)
	}
	scenario.commands = credentialIsolationCommandsFromRelease(t, scenario.release)
	hosts := make([]string, 0, len(scenario.commands))
	for _, command := range scenario.commands {
		hosts = append(hosts, command.target.Hostname())
		scenario.secrets = append(scenario.secrets, newCredentialIsolationSecret(t, command.field))
	}
	scenario.network = newCredentialIsolationProviderNetwork(t, hosts)
	// These are the connector setup values dexcli dev passes to Dex Web for
	// --connector-config-dir and --connector-release-override.
	setup, err := newConnectorSetup(&Config{
		BindAddress: "127.0.0.1", ConnectorSetupEnabled: true, ConnectorConfigDirectory: t.TempDir(),
		ConnectorReleaseOverrides: map[string]string{scenario.release.ConnectorID: releaseDirectory},
	}, connectorTestDefinitionProvider(t, []connectorDefinitionIdentity{credentialIsolationDefinitionIdentity(t, scenario.release)}))
	if err != nil {
		t.Fatalf("create Dex Web connector setup: %v", err)
	}
	scenario.network.redirectProviderClient(t, setup.providerHTTPClient)
	mux := http.NewServeMux()
	setup.registerHandlers(mux)
	scenario.webServer = httptest.NewServer(mux)
	t.Cleanup(scenario.webServer.Close)
	return scenario
}

func (scenario *credentialIsolationScenario) requireConnectionStatus(t *testing.T, expectedStatus string) {
	t.Helper()
	status, body := scenario.requestDexWeb(t, http.MethodGet, "/api/v2/connector-connections", nil)
	var catalog struct {
		CSRFToken          string                              `json:"csrfToken"`
		DefinitionRevision string                              `json:"definitionRevision"`
		Connections        []credentialIsolationConnectionView `json:"connections"`
	}
	if status != http.StatusOK || json.Unmarshal(body, &catalog) != nil {
		t.Fatalf("list connections: status=%d body=%s", status, body)
	}
	scenario.csrfToken = catalog.CSRFToken
	scenario.definitionRevision = catalog.DefinitionRevision
	for _, connection := range catalog.Connections {
		if connection.ConnectorID != scenario.release.ConnectorID || connection.ConnectionName != credentialIsolationConnectionName {
			continue
		}
		if connection.Status != expectedStatus || !connection.LocalOverride || len(connection.Uses) != 1 ||
			len(connection.Uses[0].ConfigurationUI.Units) != 1 {
			t.Fatalf("connection view = %+v, want status %s from the local release override", connection, expectedStatus)
		}
		scenario.connection = connection
		return
	}
	t.Fatalf("connection %s/%s is missing from %s", scenario.release.ConnectorID, credentialIsolationConnectionName, body)
}

func (scenario *credentialIsolationScenario) saveCredentials(t *testing.T, credentials map[string]any) {
	t.Helper()
	path := "/api/v2/connector-connections/" + url.PathEscape(scenario.release.ConnectorID) + "/" + url.PathEscape(credentialIsolationConnectionName)
	status, body := scenario.requestDexWeb(t, http.MethodPut, path, map[string]any{
		"modulePath": scenario.release.ModulePath, "moduleVersion": scenario.release.Version,
		"provider": scenario.release.Manifest.Spec.Provider, "configuration": map[string]any{},
		"credentials": credentials, "credentialExpiresAt": nil,
	})
	if status != http.StatusOK {
		t.Fatalf("save credentials: status=%d body=%s", status, body)
	}
}

func (scenario *credentialIsolationScenario) allCredentials() map[string]any {
	credentials := make(map[string]any, len(scenario.secrets))
	for _, secret := range scenario.secrets {
		credentials[secret.field] = secret.value
	}
	return credentials
}

func (scenario *credentialIsolationScenario) credentialsWithUnusableField(field string, unusable credentialIsolationUnusableValue) map[string]any {
	credentials := scenario.allCredentials()
	delete(credentials, field)
	if !unusable.isOmitted {
		credentials[field] = unusable.value
	}
	return credentials
}

func (scenario *credentialIsolationScenario) openStudioSession(t *testing.T) {
	t.Helper()
	status, body := scenario.requestDexWeb(t, http.MethodPost, "/api/v2/connector-ui-sessions", map[string]string{
		"connectorId": scenario.release.ConnectorID, "connectionName": credentialIsolationConnectionName,
	})
	var session struct {
		SessionNonce  string `json:"sessionNonce"`
		EntrypointURL string `json:"entrypointUrl"`
		Manifest      struct {
			Spec struct {
				Studio struct {
					Setup struct {
						BackendCapabilities []string `json:"backendCapabilities"`
					} `json:"setup"`
					Commands []struct {
						ID         string `json:"id"`
						Capability string `json:"capability"`
					} `json:"commands"`
				} `json:"studio"`
			} `json:"spec"`
		} `json:"manifest"`
	}
	if status != http.StatusOK || json.Unmarshal(body, &session) != nil || session.SessionNonce == "" || session.EntrypointURL == "" {
		t.Fatalf("create UI session: status=%d body=%s", status, body)
	}
	var sessionCommandIDs, fixtureCommandIDs []string
	for _, command := range session.Manifest.Spec.Studio.Commands {
		sessionCommandIDs = append(sessionCommandIDs, command.ID)
	}
	for _, command := range scenario.commands {
		fixtureCommandIDs = append(fixtureCommandIDs, command.id)
	}
	if !reflect.DeepEqual(sessionCommandIDs, fixtureCommandIDs) {
		t.Fatalf("UI session commands = %v, want %v", sessionCommandIDs, fixtureCommandIDs)
	}
	scenario.sessionNonce = session.SessionNonce
	if status, body := scenario.requestDexWeb(t, http.MethodGet, session.EntrypointURL, nil); status != http.StatusOK {
		t.Fatalf("load Studio entrypoint: status=%d body=%s", status, body)
	}
	scenario.requireFrameMessageOmitsSecrets(t, "connector.host.ready", scenario.hostReadyFrameMessage(session.Manifest.Spec.Studio.Setup.BackendCapabilities))
}

func (scenario *credentialIsolationScenario) requireCommandReachesOnlyItsHost(t *testing.T, command credentialIsolationCommand) {
	t.Helper()
	mark := scenario.network.markTraffic()
	status, value := scenario.executeCommand(t, command)
	if status != http.StatusOK {
		t.Fatalf("%s: status=%d response=%v", command.id, status, value)
	}
	encodedValue, err := json.Marshal(value)
	if err != nil {
		t.Fatal(err)
	}
	for _, other := range scenario.commands {
		isServedByOther := strings.Contains(string(encodedValue), credentialIsolationModelID(other.target.Hostname()))
		if isServedByOther != (other.id == command.id) {
			t.Fatalf("%s response = %s, want only the model list from %s", command.id, encodedValue, command.target.Hostname())
		}
	}
	scenario.requireOneRequestOnlyToOwnHost(t, command, mark)
}

// requireProviderFailureReturnsNeitherKey stores both keys, then makes the
// command's own host fail after Dex Web sent its key.
func (scenario *credentialIsolationScenario) requireProviderFailureReturnsNeitherKey(t *testing.T, command credentialIsolationCommand, failure credentialIsolationProviderFailure) {
	t.Helper()
	scenario.saveCredentials(t, scenario.allCredentials())
	scenario.requireConnectionStatus(t, "Ready")
	scenario.network.providers[command.target.Hostname()].failRequestsUntilCleanup(t, failure.behavior)
	mark := scenario.network.markTraffic()
	status, value := scenario.executeCommand(t, command)
	if status != http.StatusBadGateway || value["code"] != "CONNECTOR_PROVIDER_COMMAND_FAILED" {
		t.Fatalf("%s with provider %s: status=%d response=%v", command.id, failure.name, status, value)
	}
	scenario.requireOneRequestOnlyToOwnHost(t, command, mark)
}

// requireUnusableFieldSendsNothing stores field as unusable, then requires its
// command to send nothing while every other command still works.
func (scenario *credentialIsolationScenario) requireUnusableFieldSendsNothing(t *testing.T, command credentialIsolationCommand, unusable credentialIsolationUnusableValue) {
	t.Helper()
	scenario.saveCredentials(t, scenario.credentialsWithUnusableField(command.field, unusable))
	scenario.requireConnectionStatus(t, "Ready")
	scenario.requireCommandSendsNothing(t, command)
	for _, other := range scenario.commands {
		if other.id != command.id {
			scenario.requireCommandReachesOnlyItsHost(t, other)
		}
	}
}

func (scenario *credentialIsolationScenario) requireCommandSendsNothing(t *testing.T, command credentialIsolationCommand) {
	t.Helper()
	mark := scenario.network.markTraffic()
	status, value := scenario.executeCommand(t, command)
	if status != http.StatusBadGateway || value["code"] != "CONNECTOR_PROVIDER_COMMAND_FAILED" {
		t.Fatalf("%s without a usable %s: status=%d response=%v", command.id, command.field, status, value)
	}
	if dials := scenario.network.dialsSince(mark); len(dials) != 0 {
		t.Fatalf("%s without a usable %s dialed %v", command.id, command.field, dials)
	}
	for _, host := range scenario.network.hosts() {
		if requests := scenario.network.providers[host].requestsSince(mark.requestCounts[host]); len(requests) != 0 {
			t.Fatalf("%s without a usable %s sent %d requests to %s", command.id, command.field, len(requests), host)
		}
	}
}

// executeCommand runs a frame's provider.command.execute as the host page does
// and checks the result message it posts back.
func (scenario *credentialIsolationScenario) executeCommand(t *testing.T, command credentialIsolationCommand) (int, map[string]any) {
	t.Helper()
	path := "/api/v2/connector-ui-sessions/" + url.PathEscape(scenario.sessionNonce) + "/commands/" + url.PathEscape(command.id)
	status, body := scenario.requestDexWeb(t, http.MethodPost, path, map[string]any{"parameters": map[string]string{}})
	var value map[string]any
	if err := json.Unmarshal(body, &value); err != nil {
		t.Fatalf("%s: status=%d body=%s is not a JSON object: %v", command.id, status, body, err)
	}
	scenario.frameRequestCount++
	message := map[string]any{
		"type": "connector.command.result", "protocolVersion": credentialIsolationFrameProtocol,
		"sessionNonce": scenario.sessionNonce, "connectorId": scenario.release.ConnectorID,
		"requestId": fmt.Sprintf("request-%d", scenario.frameRequestCount), "ok": status >= 200 && status < 300,
	}
	if status >= 200 && status < 300 {
		message["value"] = value
	} else {
		// The host page forwards readResponseJSON's error message, the response's error field.
		errorText, _ := value["error"].(string)
		message["error"] = map[string]any{"code": "COMMAND_FAILED", "message": strings.TrimSpace(errorText)}
	}
	scenario.requireFrameMessageOmitsSecrets(t, "connector.command.result for "+command.id, message)
	return status, value
}

// hostReadyFrameMessage builds the host page's connector.host.ready message; its
// target value is the whole use configuration.
func (scenario *credentialIsolationScenario) hostReadyFrameMessage(setupCapabilities []string) map[string]any {
	supported := map[string]bool{"oauth.connection.manage": true, "use.configuration.write": true}
	for _, command := range scenario.commands {
		supported[command.capability] = true
	}
	capabilities := []string{}
	for _, capability := range setupCapabilities {
		if supported[capability] {
			capabilities = append(capabilities, capability)
		}
	}
	state := map[string]string{"Ready": "connected", "Expired": "expired", "Missing": "not_configured"}[scenario.connection.Status]
	if state == "" {
		state = "error"
	}
	use := scenario.connection.Uses[0]
	unit := use.ConfigurationUI.Units[0]
	return map[string]any{
		"type": "connector.host.ready", "protocolVersion": credentialIsolationFrameProtocol,
		"sessionNonce": scenario.sessionNonce, "connectorId": scenario.release.ConnectorID,
		"capabilities": capabilities,
		"connection":   map[string]any{"state": state, "grantedScopes": []string{}, "detail": scenario.connection.Status},
		"target": map[string]any{
			"kind":       "configurationUnit",
			"scope":      map[string]any{"kind": "operation", "operationId": use.OperationID, "flowType": use.FlowName, "stepType": use.StepName},
			"instanceId": unit.ID, "unitId": unit.UnitID, "label": unit.Label, "description": unit.Description,
			"required": unit.Required, "bindings": unit.Bindings, "value": use.Configuration,
		},
	}
}

func (scenario *credentialIsolationScenario) requireOneRequestOnlyToOwnHost(t *testing.T, command credentialIsolationCommand, mark credentialIsolationTrafficMark) {
	t.Helper()
	if dials := scenario.network.dialsSince(mark); !reflect.DeepEqual(dials, []string{command.target.Host + ":443"}) {
		t.Fatalf("%s dialed %v, want only %s:443", command.id, dials, command.target.Host)
	}
	for _, host := range scenario.network.hosts() {
		requests := scenario.network.providers[host].requestsSince(mark.requestCounts[host])
		if host != command.target.Hostname() {
			if len(requests) != 0 {
				t.Fatalf("%s sent %d requests to %s", command.id, len(requests), host)
			}
			continue
		}
		if len(requests) != 1 {
			t.Fatalf("%s sent %d requests to %s, want 1", command.id, len(requests), host)
		}
		scenario.requireProviderRequest(t, command, requests[0])
	}
}

func (scenario *credentialIsolationScenario) requireProviderRequest(t *testing.T, command credentialIsolationCommand, request credentialIsolationProviderRequest) {
	t.Helper()
	host := command.target.Hostname()
	query, err := url.ParseQuery(request.rawQuery)
	if err != nil || request.method != http.MethodGet || request.host != host || request.serverName != host ||
		request.path != command.target.Path || !reflect.DeepEqual(query, command.fixedQuery) || request.body != "" {
		t.Fatalf("%s request = %s %s%s?%s (TLS server name %q, %d body bytes), want GET %s%s?%s",
			command.id, request.method, request.host, request.path, request.rawQuery, request.serverName, len(request.body),
			host, command.target.Path, command.fixedQuery.Encode())
	}
	ownSecret := scenario.secretForField(t, command.field)
	if request.header.Get("Authorization") != "Bearer "+ownSecret.value || len(request.header.Values("Authorization")) != 1 {
		t.Fatalf("%s did not send exactly its own %s as the bearer credential", command.id, command.field)
	}
	for name, values := range request.header {
		if name != "Authorization" && strings.Contains(strings.Join(values, "\n"), ownSecret.marker) {
			t.Fatalf("%s sent its %s in the %s header", command.id, command.field, name)
		}
	}
	requestText := request.path + "?" + request.rawQuery + "\n" + request.body
	for name, values := range request.header {
		requestText += "\n" + name + ": " + strings.Join(values, ", ")
	}
	for _, secret := range scenario.secrets {
		if secret.field != command.field && strings.Contains(requestText, secret.marker) {
			t.Fatalf("%s sent %s to %s", command.id, secret.field, host)
		}
	}
}

// requestDexWeb calls Dex Web as the Connections page does and fails when the
// response holds a secret.
func (scenario *credentialIsolationScenario) requestDexWeb(t *testing.T, method string, path string, body any) (int, []byte) {
	t.Helper()
	var requestBody io.Reader
	if body != nil {
		encoded, err := json.Marshal(body)
		if err != nil {
			t.Fatal(err)
		}
		requestBody = bytes.NewReader(encoded)
	}
	request, err := http.NewRequestWithContext(t.Context(), method, scenario.webServer.URL+path, requestBody)
	if err != nil {
		t.Fatal(err)
	}
	request.Header.Set("Origin", scenario.webServer.URL)
	request.Header.Set(connectorCSRFHeader, scenario.csrfToken)
	request.Header.Set(api.V2DefinitionRevisionHeader, scenario.definitionRevision)
	if body != nil {
		request.Header.Set("Content-Type", "application/json")
	}
	response, err := scenario.webServer.Client().Do(request)
	if err != nil {
		t.Fatalf("%s %s: %v", method, path, err)
	}
	defer response.Body.Close()
	contents, err := io.ReadAll(response.Body)
	if err != nil {
		t.Fatalf("%s %s: read response: %v", method, path, err)
	}
	responseText := response.Status + "\n" + string(contents)
	for name, values := range response.Header {
		responseText += "\n" + name + ": " + strings.Join(values, ", ")
	}
	scenario.requireTextOmitsSecrets(t, method+" "+path+" response", responseText)
	return response.StatusCode, contents
}

func (scenario *credentialIsolationScenario) requireFrameMessageOmitsSecrets(t *testing.T, description string, message map[string]any) {
	t.Helper()
	encoded, err := json.Marshal(message)
	if err != nil {
		t.Fatalf("encode %s frame message: %v", description, err)
	}
	scenario.requireTextOmitsSecrets(t, description+" frame message", string(encoded))
}

func (scenario *credentialIsolationScenario) requireTextOmitsSecrets(t *testing.T, description string, text string) {
	t.Helper()
	for _, secret := range scenario.secrets {
		if strings.Contains(text, secret.marker) {
			t.Fatalf("%s contains the %s secret", description, secret.field)
		}
	}
}

func (scenario *credentialIsolationScenario) secretForField(t *testing.T, field string) credentialIsolationSecret {
	t.Helper()
	for _, secret := range scenario.secrets {
		if secret.field == field {
			return secret
		}
	}
	t.Fatalf("no secret for %s", field)
	return credentialIsolationSecret{}
}

// credentialIsolationCommandsFromRelease requires each fixture command to bind
// its own optional secret field and HTTPS host.
func credentialIsolationCommandsFromRelease(t *testing.T, release credentialIsolationRelease) []credentialIsolationCommand {
	t.Helper()
	studio := release.Manifest.Spec.Studio
	if studio == nil || len(studio.Commands) < 2 {
		t.Fatal("credential isolation fixture must declare at least two Studio commands")
	}
	optionalSecretFields := make(map[string]bool)
	for _, field := range release.Manifest.Spec.Auth.Fields {
		if field.Type == "secretString" && !field.Required {
			optionalSecretFields[field.Name] = true
		}
	}
	boundFields := make(map[string]bool)
	boundHosts := make(map[string]bool)
	commands := make([]credentialIsolationCommand, 0, len(studio.Commands))
	for _, declared := range studio.Commands {
		request := declared.Request
		target, err := url.Parse(request.URL)
		// cli-v0.13.8 sends only the bearer scheme and drops fixed headers.
		if err != nil || target.Scheme != "https" || target.Port() != "" || request.Method != http.MethodGet ||
			request.Credential.Scheme != "bearer" || len(request.FixedHeaders) != 0 || len(request.Parameters) != 0 ||
			!optionalSecretFields[request.Credential.Field] || boundFields[request.Credential.Field] || boundHosts[target.Hostname()] {
			t.Fatalf("fixture command %s must be a parameterless bearer GET to its own HTTPS host with its own optional secret field", declared.ID)
		}
		boundFields[request.Credential.Field] = true
		boundHosts[target.Hostname()] = true
		fixedQuery := url.Values{}
		for name, value := range request.FixedQuery {
			fixedQuery.Set(name, value)
		}
		commands = append(commands, credentialIsolationCommand{
			id: declared.ID, capability: declared.Capability, field: request.Credential.Field,
			target: target, fixedQuery: fixedQuery,
		})
	}
	return commands
}

// credentialIsolationProviderNetwork routes each provider host to a fake HTTPS
// server whose certificate names only that host.
type credentialIsolationProviderNetwork struct {
	providers map[string]*credentialIsolationProvider
	roots     *x509.CertPool
	mu        sync.Mutex
	dials     []string
}

type credentialIsolationTrafficMark struct {
	dialCount     int
	requestCounts map[string]int
}

func newCredentialIsolationProviderNetwork(t *testing.T, hosts []string) *credentialIsolationProviderNetwork {
	t.Helper()
	network := &credentialIsolationProviderNetwork{
		providers: make(map[string]*credentialIsolationProvider, len(hosts)), roots: x509.NewCertPool(),
	}
	for _, host := range hosts {
		provider := newCredentialIsolationProvider(t, host)
		network.providers[host] = provider
		network.roots.AddCert(provider.certificate)
	}
	return network
}

// redirectProviderClient keeps Dex Web's timeout and redirect policy, changing
// only dialing, trust roots, proxy, and connection reuse.
func (network *credentialIsolationProviderNetwork) redirectProviderClient(t *testing.T, client *http.Client) {
	t.Helper()
	if client == nil {
		t.Fatal("Dex Web connector setup has no provider HTTP client")
	}
	baseTransport := client.Transport
	if baseTransport == nil {
		baseTransport = http.DefaultTransport
	}
	transport, ok := baseTransport.(*http.Transport)
	if !ok {
		t.Fatalf("Dex Web provider transport %T cannot reach the fake provider hosts", baseTransport)
	}
	redirected := transport.Clone()
	redirected.Proxy = nil
	redirected.DialTLSContext = nil
	redirected.DialContext = network.dialProvider
	// One connection per request makes every provider request visible as a dial.
	redirected.DisableKeepAlives = true
	redirected.TLSClientConfig = &tls.Config{RootCAs: network.roots, MinVersion: tls.VersionTLS12}
	client.Transport = redirected
}

func (network *credentialIsolationProviderNetwork) dialProvider(ctx context.Context, networkName string, address string) (net.Conn, error) {
	network.mu.Lock()
	network.dials = append(network.dials, address)
	network.mu.Unlock()
	host, port, err := net.SplitHostPort(address)
	if err != nil {
		return nil, err
	}
	provider, found := network.providers[host]
	if !found || port != "443" {
		return nil, fmt.Errorf("Dex Web dialed undeclared provider address %s", address)
	}
	var dialer net.Dialer
	return dialer.DialContext(ctx, networkName, provider.server.Listener.Addr().String())
}

func (network *credentialIsolationProviderNetwork) markTraffic() credentialIsolationTrafficMark {
	network.mu.Lock()
	mark := credentialIsolationTrafficMark{dialCount: len(network.dials), requestCounts: make(map[string]int, len(network.providers))}
	network.mu.Unlock()
	for host, provider := range network.providers {
		mark.requestCounts[host] = provider.requestCount()
	}
	return mark
}

func (network *credentialIsolationProviderNetwork) dialsSince(mark credentialIsolationTrafficMark) []string {
	network.mu.Lock()
	defer network.mu.Unlock()
	return append([]string(nil), network.dials[mark.dialCount:]...)
}

func (network *credentialIsolationProviderNetwork) hosts() []string {
	hosts := make([]string, 0, len(network.providers))
	for host := range network.providers {
		hosts = append(hosts, host)
	}
	sort.Strings(hosts)
	return hosts
}

type credentialIsolationProvider struct {
	host        string
	certificate *x509.Certificate
	server      *httptest.Server
	mu          sync.Mutex
	requests    []credentialIsolationProviderRequest
	behavior    credentialIsolationProviderBehavior
}

type credentialIsolationProviderRequest struct {
	method     string
	host       string
	serverName string
	path       string
	rawQuery   string
	header     http.Header
	body       string
}

func newCredentialIsolationProvider(t *testing.T, host string) *credentialIsolationProvider {
	t.Helper()
	certificate := newCredentialIsolationCertificate(t, host)
	provider := &credentialIsolationProvider{host: host, certificate: certificate.Leaf}
	provider.server = httptest.NewUnstartedServer(http.HandlerFunc(provider.serveProviderRequest))
	provider.server.TLS = &tls.Config{Certificates: []tls.Certificate{certificate}, MinVersion: tls.VersionTLS12}
	provider.server.StartTLS()
	t.Cleanup(provider.server.Close)
	return provider
}

// failRequestsUntilCleanup answers every request with behavior until t finishes.
func (provider *credentialIsolationProvider) failRequestsUntilCleanup(t *testing.T, behavior credentialIsolationProviderBehavior) {
	t.Helper()
	provider.mu.Lock()
	provider.behavior = behavior
	provider.mu.Unlock()
	t.Cleanup(func() {
		provider.mu.Lock()
		provider.behavior = credentialIsolationProviderListsModels
		provider.mu.Unlock()
	})
}

func (provider *credentialIsolationProvider) serveProviderRequest(response http.ResponseWriter, request *http.Request) {
	body, readErr := io.ReadAll(request.Body)
	recorded := credentialIsolationProviderRequest{
		method: request.Method, host: request.Host, path: request.URL.Path, rawQuery: request.URL.RawQuery,
		header: request.Header.Clone(), body: string(body),
	}
	if request.TLS != nil {
		recorded.serverName = request.TLS.ServerName
	}
	provider.mu.Lock()
	provider.requests = append(provider.requests, recorded)
	behavior := provider.behavior
	provider.mu.Unlock()
	if readErr != nil {
		http.Error(response, "request body is unreadable", http.StatusBadRequest)
		return
	}
	authorization := request.Header.Get("Authorization")
	models := []map[string]string{{"id": credentialIsolationModelID(provider.host)}}
	switch behavior {
	case credentialIsolationProviderRejectsEchoingAuthorization:
		writeCredentialIsolationProviderJSON(response, http.StatusUnauthorized, map[string]any{
			"error": map[string]string{"type": "authentication_error", "message": "Incorrect API key provided: " + authorization},
		})
	case credentialIsolationProviderListsModelsEchoingKey:
		models[0]["owned_by"] = strings.TrimPrefix(authorization, "Bearer ")
		writeCredentialIsolationProviderJSON(response, http.StatusOK, map[string]any{"object": "list", "data": models})
	case credentialIsolationProviderClosesConnection:
		// Aborting before any write closes the connection without a response.
		panic(http.ErrAbortHandler)
	default:
		writeCredentialIsolationProviderJSON(response, http.StatusOK, map[string]any{"object": "list", "data": models})
	}
}

func (provider *credentialIsolationProvider) requestCount() int {
	provider.mu.Lock()
	defer provider.mu.Unlock()
	return len(provider.requests)
}

func (provider *credentialIsolationProvider) requestsSince(count int) []credentialIsolationProviderRequest {
	provider.mu.Lock()
	defer provider.mu.Unlock()
	return append([]credentialIsolationProviderRequest(nil), provider.requests[count:]...)
}

func writeCredentialIsolationProviderJSON(response http.ResponseWriter, status int, value map[string]any) {
	response.Header().Set("Content-Type", "application/json")
	response.WriteHeader(status)
	// A failed write truncates the body, which Dex Web rejects and the test reports.
	_ = json.NewEncoder(response).Encode(value)
}

// newCredentialIsolationCertificate returns a self-signed certificate valid only for host.
func newCredentialIsolationCertificate(t *testing.T, host string) tls.Certificate {
	t.Helper()
	key, err := ecdsa.GenerateKey(elliptic.P256(), rand.Reader)
	if err != nil {
		t.Fatal(err)
	}
	serialNumber, err := rand.Int(rand.Reader, new(big.Int).Lsh(big.NewInt(1), 62))
	if err != nil {
		t.Fatal(err)
	}
	template := &x509.Certificate{
		SerialNumber: serialNumber, Subject: pkix.Name{CommonName: host}, DNSNames: []string{host},
		NotBefore: time.Now().Add(-time.Hour), NotAfter: time.Now().Add(time.Hour),
		KeyUsage: x509.KeyUsageDigitalSignature, ExtKeyUsage: []x509.ExtKeyUsage{x509.ExtKeyUsageServerAuth},
		BasicConstraintsValid: true,
	}
	certificateDER, err := x509.CreateCertificate(rand.Reader, template, template, &key.PublicKey, key)
	if err != nil {
		t.Fatal(err)
	}
	leaf, err := x509.ParseCertificate(certificateDER)
	if err != nil {
		t.Fatal(err)
	}
	return tls.Certificate{Certificate: [][]byte{certificateDER}, PrivateKey: key, Leaf: leaf}
}
