const (
	connectionRecordReleasesEnvironment = "DEX_CONNECTOR_COMPAT_CONNECTION_RECORD_RELEASES"
	connectionRecordOutputEnvironment   = "DEX_CONNECTOR_COMPAT_CONNECTION_RECORDS_OUTPUT"
	connectionRecordConnectionName      = "contract"
)

type connectionRecordFixtureRelease struct {
	ConnectorID string `json:"connectorId"`
	ModulePath  string `json:"modulePath"`
	Version     string `json:"version"`
	Manifest    struct {
		Spec struct {
			Provider   string                      `json:"provider"`
			Auth       connectionRecordFixtureAuth `json:"auth"`
			Operations []struct {
				Name string `json:"name"`
				Kind string `json:"kind"`
			} `json:"operations"`
		} `json:"spec"`
	} `json:"manifest"`
}

type connectionRecordFixtureAuth struct {
	Selection string                          `json:"selection"`
	Methods   []connectionRecordFixtureMethod `json:"methods"`
}

type connectionRecordFixtureMethod struct {
	ID     string                         `json:"id"`
	Fields []connectionRecordFixtureField `json:"fields"`
}

type connectionRecordFixtureField struct {
	Name     string `json:"name"`
	Required bool   `json:"required"`
}

type connectionRecordContractClient struct {
	server             *httptest.Server
	csrfToken          string
	definitionRevision string
}

// TestConnectorConnectionRecordContract writes one connection per authentication shape and one use configuration.
func TestConnectorConnectionRecordContract(t *testing.T) {
	outputDirectory := os.Getenv(connectionRecordOutputEnvironment)
	if outputDirectory == "" {
		t.Fatal(connectionRecordOutputEnvironment + " is required")
	}
	isolationDirectory := os.Getenv(credentialIsolationReleaseEnvironment)
	if isolationDirectory == "" {
		t.Fatal(credentialIsolationReleaseEnvironment + " is required")
	}
	var isolationRelease credentialIsolationRelease
	readConnectionRecordRelease(t, isolationDirectory, &isolationRelease)
	isolationIdentity := credentialIsolationDefinitionIdentity(t, isolationRelease)
	identities := []connectorDefinitionIdentity{isolationIdentity}
	overrides := map[string]string{isolationRelease.ConnectorID: isolationDirectory}
	var fixtures []connectionRecordFixtureRelease
	for _, directory := range filepath.SplitList(os.Getenv(connectionRecordReleasesEnvironment)) {
		var fixture connectionRecordFixtureRelease
		readConnectionRecordRelease(t, directory, &fixture)
		if len(fixture.Manifest.Spec.Operations) == 0 {
			t.Fatalf("connection record fixture %s declares no operation", fixture.ConnectorID)
		}
		fixtures = append(fixtures, fixture)
		overrides[fixture.ConnectorID] = directory
		identities = append(identities, connectorDefinitionIdentity{
			ConnectorID: fixture.ConnectorID, ModulePath: fixture.ModulePath, ModuleVersion: fixture.Version,
			ConnectionName: connectionRecordConnectionName, OperationID: fixture.Manifest.Spec.Operations[0].Name,
			OperationKind: fixture.Manifest.Spec.Operations[0].Kind, ConfigurationEnabled: true,
		})
	}
	if len(fixtures) < 2 {
		t.Fatalf("%s must list the single-method and multiple-method fixture releases", connectionRecordReleasesEnvironment)
	}
	configurationDirectory := t.TempDir()
	setup, err := newConnectorSetup(&Config{
		BindAddress: "127.0.0.1", ConnectorSetupEnabled: true, ConnectorConfigDirectory: configurationDirectory,
		ConnectorReleaseOverrides: overrides,
	}, connectorTestDefinitionProvider(t, identities))
	if err != nil {
		t.Fatalf("create Dex Web connector setup: %v", err)
	}
	mux := http.NewServeMux()
	setup.registerHandlers(mux)
	client := &connectionRecordContractClient{server: httptest.NewServer(mux)}
	defer client.server.Close()
	client.loadSession(t)

	client.putConnection(t, isolationRelease.ConnectorID, credentialIsolationConnectionName, map[string]any{
		"modulePath": isolationRelease.ModulePath, "moduleVersion": isolationRelease.Version,
		"provider": isolationRelease.Manifest.Spec.Provider, "configuration": map[string]any{},
		"credentials": map[string]any{"alpha_api_key": "contract-alpha-secret", "beta_api_key": "contract-beta-secret"},
		"credentialExpiresAt": nil,
	})
	for _, fixture := range fixtures {
		if fixture.Manifest.Spec.Auth.Selection == "multiple" && !supportsMultipleAuthMethods() {
			t.Logf("this Dex Web release cannot save multiple authentication methods; fixture %s activates once it can", fixture.ConnectorID)
			continue
		}
		client.putConnection(t, fixture.ConnectorID, connectionRecordConnectionName, connectionRecordWriteBody(t, fixture))
	}
	useConfigurationPath := "/api/v2/connector-use-configurations/" + url.PathEscape(isolationRelease.ConnectorID) + "/" +
		url.PathEscape(credentialIsolationConnectionName) + "/" + url.PathEscape(isolationIdentity.OperationID) + "/" +
		url.PathEscape(client.isolationUse(t, isolationRelease.ConnectorID, "flowName")) + "/" +
		url.PathEscape(client.isolationUse(t, isolationRelease.ConnectorID, "stepName"))
	port := isolationIdentity.ConfigurationUI.Units[0].Bindings[0].Port
	client.request(t, http.MethodPut, useConfigurationPath, map[string]any{"configuration": map[string]any{port: "contract-model"}})

	if err := os.MkdirAll(outputDirectory, 0o700); err != nil {
		t.Fatal(err)
	}
	for _, name := range []string{connectorConnectionsFileName, connectorUseConfigurationsFileName} {
		contents, err := os.ReadFile(filepath.Join(configurationDirectory, name))
		if err != nil {
			t.Fatalf("Dex Web did not write %s: %v", name, err)
		}
		if err := os.WriteFile(filepath.Join(outputDirectory, name), contents, 0o600); err != nil {
			t.Fatal(err)
		}
	}
}

// connectionRecordWriteBody selects the non-default method, or every method in reverse order, to exercise explicit choices.
func connectionRecordWriteBody(t *testing.T, fixture connectionRecordFixtureRelease) map[string]any {
	t.Helper()
	auth := fixture.Manifest.Spec.Auth
	body := map[string]any{
		"modulePath": fixture.ModulePath, "moduleVersion": fixture.Version, "provider": fixture.Manifest.Spec.Provider,
		"configuration": map[string]any{}, "credentialExpiresAt": nil,
	}
	credentials := map[string]any{}
	addCredentials := func(fields []connectionRecordFixtureField) {
		for _, field := range fields {
			if field.Required {
				credentials[field.Name] = "contract-" + field.Name
			}
		}
	}
	switch {
	case auth.Selection == "multiple":
		methodIDs := make([]string, 0, len(auth.Methods))
		for _, method := range slices.Backward(auth.Methods) {
			methodIDs = append(methodIDs, method.ID)
			addCredentials(method.Fields)
		}
		body["authMethodIds"] = methodIDs
	case len(auth.Methods) > 1:
		method := auth.Methods[len(auth.Methods)-1]
		body["authMethodId"] = method.ID
		addCredentials(method.Fields)
	default:
		t.Fatalf("connection record fixture %s must declare named authentication methods", fixture.ConnectorID)
	}
	body["credentials"] = credentials
	return body
}

// supportsMultipleAuthMethods reports whether this Dex Web release stores multiple selected methods.
func supportsMultipleAuthMethods() bool {
	_, found := reflect.TypeOf(localConnectorConnection{}).FieldByName("AuthMethodIDs")
	return found
}

func readConnectionRecordRelease(t *testing.T, directory string, destination any) {
	t.Helper()
	metadata, err := os.ReadFile(filepath.Join(directory, connectorReleaseMetadataName))
	if err != nil {
		t.Fatal(err)
	}
	if err := json.Unmarshal(metadata, destination); err != nil {
		t.Fatalf("decode %s: %v", directory, err)
	}
}

func (client *connectionRecordContractClient) loadSession(t *testing.T) {
	t.Helper()
	var catalog struct {
		CSRFToken          string `json:"csrfToken"`
		DefinitionRevision string `json:"definitionRevision"`
	}
	if err := json.Unmarshal(client.request(t, http.MethodGet, "/api/v2/connector-connections", nil), &catalog); err != nil {
		t.Fatalf("decode connection catalog: %v", err)
	}
	client.csrfToken = catalog.CSRFToken
	client.definitionRevision = catalog.DefinitionRevision
}

func (client *connectionRecordContractClient) putConnection(t *testing.T, connectorID string, connectionName string, body map[string]any) {
	t.Helper()
	client.request(t, http.MethodPut, "/api/v2/connector-connections/"+url.PathEscape(connectorID)+"/"+url.PathEscape(connectionName), body)
}

// isolationUse returns one member of the isolation connection's only Step use.
func (client *connectionRecordContractClient) isolationUse(t *testing.T, connectorID string, member string) string {
	t.Helper()
	contents := client.request(t, http.MethodGet, "/api/v2/connector-connections", nil)
	var raw struct {
		Connections []struct {
			ConnectorID    string                       `json:"connectorId"`
			ConnectionName string                       `json:"connectionName"`
			Uses           []map[string]json.RawMessage `json:"uses"`
		} `json:"connections"`
	}
	if err := json.Unmarshal(contents, &raw); err != nil {
		t.Fatalf("decode connection catalog: %v", err)
	}
	for _, connection := range raw.Connections {
		if connection.ConnectorID != connectorID || connection.ConnectionName != credentialIsolationConnectionName || len(connection.Uses) != 1 {
			continue
		}
		var value string
		if err := json.Unmarshal(connection.Uses[0][member], &value); err != nil || value == "" {
			t.Fatalf("isolation use has no %s: %s", member, contents)
		}
		return value
	}
	t.Fatalf("isolation connection use is missing from %s", contents)
	return ""
}

func (client *connectionRecordContractClient) request(t *testing.T, method string, path string, body any) []byte {
	t.Helper()
	var requestBody io.Reader
	if body != nil {
		encoded, err := json.Marshal(body)
		if err != nil {
			t.Fatal(err)
		}
		requestBody = bytes.NewReader(encoded)
	}
	request, err := http.NewRequestWithContext(t.Context(), method, client.server.URL+path, requestBody)
	if err != nil {
		t.Fatal(err)
	}
	request.Header.Set("Origin", client.server.URL)
	request.Header.Set(connectorCSRFHeader, client.csrfToken)
	request.Header.Set(api.V2DefinitionRevisionHeader, client.definitionRevision)
	if body != nil {
		request.Header.Set("Content-Type", "application/json")
	}
	response, err := client.server.Client().Do(request)
	if err != nil {
		t.Fatalf("%s %s: %v", method, path, err)
	}
	defer response.Body.Close()
	contents, err := io.ReadAll(response.Body)
	if err != nil {
		t.Fatal(err)
	}
	if response.StatusCode != http.StatusOK {
		t.Fatalf("%s %s: status=%d body=%s", method, path, response.StatusCode, contents)
	}
	return contents
}
