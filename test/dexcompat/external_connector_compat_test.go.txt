func TestExternalConnectorCompatibility(t *testing.T) {
	artifactRoot := os.Getenv("DEX_CONNECTOR_COMPAT_ARTIFACT_ROOT")
	if artifactRoot == "" {
		t.Fatal("DEX_CONNECTOR_COMPAT_ARTIFACT_ROOT is required")
	}
	metadataPaths, err := filepath.Glob(filepath.Join(artifactRoot, "connectors", "*", "connector-release.json"))
	if err != nil {
		t.Fatal(err)
	}
	nestedMetadataPaths, err := filepath.Glob(filepath.Join(artifactRoot, "connectors", "*", "*", "connector-release.json"))
	if err != nil {
		t.Fatal(err)
	}
	metadataPaths = append(metadataPaths, nestedMetadataPaths...)
	if len(metadataPaths) == 0 {
		t.Fatal("no Connector release metadata found")
	}
	artifacts := make(map[string]string)
	identities := make([]connectorDefinitionIdentity, 0, len(metadataPaths))
	metadataByConnector := make(map[string][]byte, len(metadataPaths))
	for _, metadataPath := range metadataPaths {
		metadata, readErr := os.ReadFile(metadataPath)
		if readErr != nil {
			t.Fatal(readErr)
		}
		var release connectorRelease
		if unmarshalErr := json.Unmarshal(metadata, &release); unmarshalErr != nil {
			t.Fatalf("decode %s: %v", metadataPath, unmarshalErr)
		}
		operationID, operationKind := externalFirstOperation(metadata)
		identity := connectorDefinitionIdentity{
			ConnectorID: release.ConnectorID, ModulePath: release.ModulePath, ModuleVersion: release.Version,
			ConnectionName: "compatibility-connection", OperationID: operationID, OperationKind: operationKind,
			ConfigurationEnabled: true,
		}
		identities = append(identities, identity)
		metadataByConnector[release.ConnectorID] = metadata
		base := filepath.Dir(metadataPath)
		artifacts[release.Tag+"/"+connectorReleaseMetadataName] = metadataPath
		artifacts[release.Tag+"/"+connectorReleaseDigestName] = filepath.Join(base, connectorReleaseDigestName)
		if release.UI != nil {
			artifacts[release.Tag+"/"+release.UI.Artifact] = filepath.Join(base, release.UI.Artifact)
		}
		externalValidateAuth(t, release, externalOAuthDeclarations(t, metadata))
	}
	server := httptest.NewServer(http.HandlerFunc(func(response http.ResponseWriter, request *http.Request) {
		path := strings.TrimPrefix(request.URL.Path, "/")
		artifact, found := artifacts[path]
		if !found {
			http.NotFound(response, request)
			return
		}
		http.ServeFile(response, request, artifact)
	}))
	defer server.Close()
	resolver := &connectorReleaseResolver{baseURL: server.URL, artifactRoot: filepath.Join(t.TempDir(), "artifacts"), httpClient: server.Client()}
	for _, identity := range identities {
		resolved, resolveErr := resolver.resolve(context.Background(), identity)
		if resolveErr != nil {
			t.Fatalf("resolve %s: %v", identity.ConnectorID, resolveErr)
		}
		declarations := externalOAuthDeclarations(t, metadataByConnector[identity.ConnectorID])
		if resolved.release.UI == nil {
			externalTestOAuthStart(t, resolver, identity, resolved.release, declarations)
			continue
		}
		entrypoint := filepath.Join(resolved.uiRoot, resolved.release.UI.Entrypoint)
		if _, statErr := os.Stat(entrypoint); statErr != nil {
			t.Fatalf("UI entrypoint %s: %v", identity.ConnectorID, statErr)
		}
		response := httptest.NewRecorder()
		serveConnectorUIAsset(response, httptest.NewRequest(http.MethodGet, "/connector-ui", nil), resolved.release.UI.Entrypoint, resolved.uiRoot, false, "'self'")
		if response.Code != http.StatusOK || !strings.Contains(response.Header().Get("Content-Security-Policy"), "connect-src 'none'") {
			t.Fatalf("UI response %s: status=%d CSP=%q", identity.ConnectorID, response.Code, response.Header().Get("Content-Security-Policy"))
		}
		externalTestOAuthStart(t, resolver, identity, resolved.release, declarations)
	}
	setup := connectorTestSetup(t, t.TempDir(), connectorTestDefinitionProvider(t, identities))
	setup.releases = resolver
	recorder := httptest.NewRecorder()
	setup.handleListConnections(recorder, httptest.NewRequest(http.MethodGet, "/api/v2/connector-connections", nil))
	if recorder.Code != http.StatusOK || strings.Contains(recorder.Body.String(), "Unsupported") || strings.Contains(recorder.Body.String(), "version conflict") {
		t.Fatalf("Connector catalog: status=%d body=%s", recorder.Code, recorder.Body.String())
	}
}

func externalTestOAuthStart(t *testing.T, resolver *connectorReleaseResolver, identity connectorDefinitionIdentity, release connectorRelease, declarations map[string]externalOAuthDeclaration) {
	t.Helper()
	for _, method := range externalAuthMethods(release.Manifest.Spec.Auth) {
		if method.Type == "oauth2" {
			externalTestOAuthMethodStart(t, resolver, identity, release, method, declarations[method.ID])
		}
	}
}

func externalTestOAuthMethodStart(t *testing.T, resolver *connectorReleaseResolver, identity connectorDefinitionIdentity, release connectorRelease, method connectorManifestAuthMethod, declaration externalOAuthDeclaration) {
	t.Helper()
	setup := connectorTestSetup(t, t.TempDir(), connectorTestDefinitionProvider(t, []connectorDefinitionIdentity{identity}))
	setup.releases = resolver
	credentialValues := make(map[string]any)
	credentialSecrets := make(map[string]string)
	mappedCredentials := map[string]bool{"access_token": true}
	if method.OAuth2.ClientIDCredential != "" {
		mappedCredentials[method.OAuth2.ClientIDCredential] = true
	}
	if method.OAuth2.ClientSecretCredential != "" {
		mappedCredentials[method.OAuth2.ClientSecretCredential] = true
	}
	for _, mapping := range method.OAuth2.CredentialMappings {
		mappedCredentials[mapping.Credential] = true
	}
	for _, derivation := range method.OAuth2.CredentialDerivations {
		mappedCredentials[derivation.Credential] = true
	}
	for _, field := range method.Fields {
		if !field.Required || mappedCredentials[field.Name] {
			continue
		}
		if field.Type == "secretString" {
			credentialSecrets[field.Name] = "compatibility-secret"
		} else {
			credentialValues[field.Name] = "compatibility-value"
		}
	}
	configuration := externalRequiredSettings(release, method)
	body, err := json.Marshal(map[string]any{
		"authMethodId": method.ID,
		"clientId": "compatibility-client", "clientSecret": "compatibility-client-secret",
		"configuration": configuration, "credentialValues": credentialValues, "credentialSecrets": credentialSecrets,
	})
	if err != nil {
		t.Fatal(err)
	}
	request := httptest.NewRequest(http.MethodPost, "/api/v2/connector-connections/"+identity.ConnectorID+"/"+identity.ConnectionName+"/oauth/start", strings.NewReader(string(body)))
	request.SetPathValue("connectorId", identity.ConnectorID)
	request.SetPathValue("connectionName", identity.ConnectionName)
	request.Host = "127.0.0.1:8802"
	request.Header.Set("Origin", "http://127.0.0.1:8802")
	request.Header.Set(connectorCSRFHeader, setup.csrfToken)
	request.Header.Set(api.V2DefinitionRevisionHeader, "sha256:test")
	response := httptest.NewRecorder()
	setup.handleOAuthStart(response, request)
	if response.Code != http.StatusOK {
		t.Fatalf("OAuth start %s: status=%d body=%s", identity.ConnectorID, response.Code, response.Body.String())
	}
	var result struct {
		AuthorizationURL string `json:"authorizationUrl"`
	}
	if err := json.Unmarshal(response.Body.Bytes(), &result); err != nil {
		t.Fatal(err)
	}
	authorizationURL, err := url.Parse(result.AuthorizationURL)
	if err != nil {
		t.Fatal(err)
	}
	oauth := method.OAuth2
	wantAuthorizationEndpoint := oauth.AuthorizationEndpoint
	if declaration.authorizationEndpointSetting != "" {
		wantAuthorizationEndpoint = configuration[declaration.authorizationEndpointSetting].(string)
	}
	wantScope := strings.Join(oauth.Scopes, " ")
	if declaration.scopesSetting != "" {
		wantScope = strings.Join(configuration[declaration.scopesSetting].([]string), " ")
	}
	if authorizationURL.Scheme+"://"+authorizationURL.Host+authorizationURL.Path != wantAuthorizationEndpoint ||
		authorizationURL.Query().Get("scope") != wantScope ||
		authorizationURL.Query().Get("user_scope") != strings.Join(oauth.UserScopes, " ") ||
		authorizationURL.Query().Get("state") == "" || authorizationURL.Query().Get("redirect_uri") == "" {
		t.Fatalf("OAuth authorization URL %s: %s", identity.ConnectorID, result.AuthorizationURL)
	}
	hasPKCE := authorizationURL.Query().Get("code_challenge") != "" && authorizationURL.Query().Get("code_challenge_method") == "S256"
	if hasPKCE != oauth.PKCE {
		t.Fatalf("OAuth PKCE %s: got=%v want=%v", identity.ConnectorID, hasPKCE, oauth.PKCE)
	}
}

func externalFirstOperation(metadata []byte) (string, string) {
	var document struct {
		Manifest struct {
			Spec struct {
				Operations []struct {
					Name string `json:"name"`
					Kind string `json:"kind"`
				} `json:"operations"`
			} `json:"spec"`
		} `json:"manifest"`
	}
	if err := json.Unmarshal(metadata, &document); err != nil || len(document.Manifest.Spec.Operations) == 0 {
		panic("Connector release has no operation")
	}
	for _, operation := range document.Manifest.Spec.Operations {
		if operation.Kind != "" {
			return operation.Name, operation.Kind
		}
	}
	panic("Connector release has no query or mutation operation")
}

func externalValidateAuth(t *testing.T, release connectorRelease, declarations map[string]externalOAuthDeclaration) {
	t.Helper()
	auth := release.Manifest.Spec.Auth
	methods := externalAuthMethods(auth)
	if len(auth.Methods) > 0 && auth.DefaultMethod == "" {
		t.Fatalf("%s multi-auth manifest has no default method", release.ConnectorID)
	}
	for _, method := range methods {
		switch method.Type {
		case "none":
			if len(method.Fields) != 0 || method.OAuth2 != nil {
				t.Fatalf("%s public auth must not declare credentials or OAuth metadata", release.ConnectorID)
			}
		case "apiKey", "serviceAccount":
			if len(method.Fields) == 0 {
				t.Fatalf("%s %s auth has no fields", release.ConnectorID, method.Type)
			}
		case "oauth2":
			declaration := declarations[method.ID]
			if method.OAuth2 == nil ||
				(method.OAuth2.AuthorizationEndpoint == "" && declaration.authorizationEndpointSetting == "") ||
				(method.OAuth2.TokenEndpoint == "" && declaration.tokenEndpointSetting == "") ||
				(len(method.OAuth2.Scopes) == 0 && declaration.scopesSetting == "") {
				t.Fatalf("%s OAuth metadata is incomplete", release.ConnectorID)
			}
		default:
			t.Fatalf("%s auth method %q has unsupported type %q", release.ConnectorID, method.ID, method.Type)
		}
	}
	encoded, err := json.Marshal(release)
	if err != nil {
		t.Fatal(err)
	}
	digest := sha256.Sum256(encoded)
	t.Logf("connector=%s tag=%s commit=%s metadata=%x", release.ConnectorID, release.Tag, release.SourceSHA, digest)
}

func externalAuthMethods(auth connectorManifestAuth) []connectorManifestAuthMethod {
	if len(auth.Methods) > 0 {
		return auth.Methods
	}
	return []connectorManifestAuthMethod{{
		Type: auth.Type, ConnectionKind: auth.ConnectionKind, Fields: auth.Fields,
		Guide: auth.Guide, OAuth2: auth.OAuth2,
	}}
}

// externalOAuthDeclaration names the configuration field behind each OAuth value that a release writes as
// {"fromSetting": name}. It reads the raw metadata, so this file compiles with every Dex Web release.
type externalOAuthDeclaration struct {
	authorizationEndpointSetting string
	tokenEndpointSetting         string
	scopesSetting                string
}

// externalOAuthDeclarations returns the declarations keyed by auth method ID; a single-method manifest uses "".
func externalOAuthDeclarations(t *testing.T, metadata []byte) map[string]externalOAuthDeclaration {
	t.Helper()
	type oauthDocument struct {
		AuthorizationEndpoint json.RawMessage `json:"authorizationEndpoint"`
		TokenEndpoint         json.RawMessage `json:"tokenEndpoint"`
		Scopes                json.RawMessage `json:"scopes"`
	}
	var document struct {
		Manifest struct {
			Spec struct {
				Auth struct {
					OAuth2  *oauthDocument `json:"oauth2"`
					Methods []struct {
						ID     string         `json:"id"`
						OAuth2 *oauthDocument `json:"oauth2"`
					} `json:"methods"`
				} `json:"auth"`
			} `json:"spec"`
		} `json:"manifest"`
	}
	if err := json.Unmarshal(metadata, &document); err != nil {
		t.Fatal(err)
	}
	reference := func(raw json.RawMessage) string {
		var setting struct {
			FromSetting string `json:"fromSetting"`
		}
		if len(raw) > 0 && raw[0] == '{' {
			if err := json.Unmarshal(raw, &setting); err != nil || setting.FromSetting == "" {
				t.Fatalf("invalid setting reference %s", raw)
			}
		}
		return setting.FromSetting
	}
	declare := func(oauth *oauthDocument) externalOAuthDeclaration {
		if oauth == nil {
			return externalOAuthDeclaration{}
		}
		return externalOAuthDeclaration{
			authorizationEndpointSetting: reference(oauth.AuthorizationEndpoint),
			tokenEndpointSetting:         reference(oauth.TokenEndpoint),
			scopesSetting:                reference(oauth.Scopes),
		}
	}
	declarations := map[string]externalOAuthDeclaration{"": declare(document.Manifest.Spec.Auth.OAuth2)}
	for _, method := range document.Manifest.Spec.Auth.Methods {
		declarations[method.ID] = declare(method.OAuth2)
	}
	return declarations
}

// externalRequiredSettings returns a sample value for every configuration field that OAuth start requires.
func externalRequiredSettings(release connectorRelease, method connectorManifestAuthMethod) map[string]any {
	fields := append([]connectorManifestField{}, release.Manifest.Spec.Configuration.Fields...)
	fields = append(fields, method.configurationFields()...)
	settings := make(map[string]any)
	for _, field := range fields {
		if !field.Required || field.Default != nil {
			continue
		}
		switch field.Type {
		case "url":
			settings[field.Name] = "https://idp.compat.example/" + field.Name
		case "stringList":
			settings[field.Name] = []string{"compat-scope"}
		case "integer":
			settings[field.Name] = 1
		case "boolean":
			settings[field.Name] = true
		case "duration":
			settings[field.Name] = "1s"
		case "enum":
			settings[field.Name] = field.Enum[0]
		case "stringMap":
			settings[field.Name] = map[string]string{}
		default:
			settings[field.Name] = "compat-value"
		}
	}
	return settings
}
