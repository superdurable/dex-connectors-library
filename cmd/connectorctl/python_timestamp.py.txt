"""Go time.Time JSON input parsing for generated connector models."""

import re
from datetime import datetime

_GO_TIME = re.compile(
    r"^([0-9]{4})-([0-9]{2})-([0-9]{2})T([0-9]{1,2}):([0-9]{2}):([0-9]{2})"
    r"(?:[.,]([0-9]+))?(Z|([+-])([0-9]{2}):([0-9]{2}))$"
)


def _parts(value: str, field_name: str) -> tuple[str, str, int, int]:
    match = _GO_TIME.fullmatch(value)
    if match is None:
        raise ValueError(f"{field_name}: invalid Go time.Time JSON timestamp")
    year, month, day, hour, minute, second, fraction, zone, sign, zone_hour, zone_minute = match.groups()
    year_number, month_number, day_number = int(year), int(month), int(day)
    is_leap = year_number % 4 == 0 and (year_number % 100 != 0 or year_number % 400 == 0)
    month_days = (31, 29 if is_leap else 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31)
    if not (1 <= month_number <= 12 and 1 <= day_number <= month_days[month_number - 1]
            and int(hour) <= 23 and int(minute) <= 59 and int(second) <= 59):
        raise ValueError(f"{field_name}: invalid Go time.Time JSON timestamp")
    offset_minutes = 0
    if zone != "Z":
        if int(zone_hour) > 24 or int(zone_minute) > 60:
            raise ValueError(f"{field_name}: invalid Go time.Time JSON offset")
        offset_minutes = (int(zone_hour) * 60 + int(zone_minute)) * (1 if sign == "+" else -1)
    return year, f"{month}-{day}T{int(hour):02d}:{minute}:{second}", offset_minutes, int((fraction or "")[:9].ljust(9, "0"))


def canonical_go_timestamp(value: str, field_name: str) -> str:
    """Parse Go's accepted JSON timestamp spellings and format like MarshalJSON."""
    year, clock, offset_minutes, nanoseconds = _parts(value, field_name)
    fraction = f"{nanoseconds:09d}".rstrip("0")
    if abs(offset_minutes) >= 24 * 60:
        return value
    offset = "Z" if offset_minutes == 0 else f"{'+' if offset_minutes > 0 else '-'}{abs(offset_minutes) // 60:02d}:{abs(offset_minutes) % 60:02d}"
    return f"{year}-{clock}{'.' + fraction if fraction else ''}{offset}"


def parse_go_datetime(value: str, field_name: str) -> datetime:
    """Decode a Go timestamp only when Python datetime preserves its value."""
    year, _, offset_minutes, nanoseconds = _parts(value, field_name)
    if int(year) == 0 or nanoseconds % 1000 or abs(offset_minutes) >= 24 * 60:
        raise ValueError(f"{field_name}: Go time.Time JSON timestamp cannot be represented exactly as datetime")
    canonical = canonical_go_timestamp(value, field_name)
    return datetime.fromisoformat(canonical.replace("Z", "+00:00"))


def validate_go_datetime(value: datetime, field_name: str) -> datetime:
    """Reject Python datetimes that cannot round-trip through Go JSON time."""
    if not isinstance(value, datetime):
        raise ValueError(f"{field_name}: expected a datetime or Go time.Time JSON timestamp")
    offset = value.utcoffset() if value.tzinfo is not None else None
    if offset is None or offset.total_seconds() % 60:
        raise ValueError(f"{field_name}: datetime requires a whole-minute timezone offset")
    return value
